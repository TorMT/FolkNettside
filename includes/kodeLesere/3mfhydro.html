document.getElementById("hydro-gz").textContent = hydroData?.gz?.toFixed(3) || '?'; } // --- Komponent-liste highlight --- let highlightedComponent = null; function buildComponentList(mesh) { const componentList = document.getElementById('componentList'); const existingChips = componentList.querySelectorAll('.component-chip:not([data-static])'); existingChips.forEach(chip => chip.remove()); // Alltid legg til standard komponenter const standardChips = [ { comp: 'hull', name: 'Skrog' }, { comp: 'wl', name: 'Vannlinje' }, { comp: 'cb', name: 'CB' }, { comp: 'cg', name: 'CG' }, { comp: 'wp', name: 'Waterplane' } ]; // Legg til dynamiske komponenter fra modellen const dynamicComponents = []; if (mesh.userData?.componentNames) { mesh.userData.componentNames.forEach(name => { const lowerName = name.toLowerCase(); let compType = 'component'; if (lowerName.includes('thruster') || lowerName.includes('propeller')) compType = 'thrusters'; else if (lowerName.includes('battery') || lowerName.includes('batt')) compType = 'batteries'; else if (lowerName.includes('sensor') || lowerName.includes('lidar') || lowerName.includes('camera')) compType = 'sensors'; else compType = name.toLowerCase().replace(/[^a-z0-9]/g, ''); dynamicComponents.push({ comp: compType, name: name }); }); } else if (mesh.children && mesh.children.length > 0) { // Fallback: travers mesh for navn mesh.traverse(child => { if (child.name && child !== mesh) { const lowerName = child.name.toLowerCase(); let compType = 'component'; if (lowerName.includes('thruster') || lowerName.includes('propeller')) compType = 'thrusters'; else if (lowerName.includes('battery') || lowerName.includes('batt')) compType = 'batteries'; else if (lowerName.includes('sensor') || lowerName.includes('lidar') || lowerName.includes('camera')) compType = 'sensors'; else compType = child.name.toLowerCase().replace(/[^a-z0-9]/g, ''); dynamicComponents.push({ comp: compType, name: child.name }); } }); } // Kombiner og fjern duplikater const allComponents = [...standardChips]; dynamicComponents.forEach(dc => { if (!allComponents.find(ac => ac.comp === dc.comp)) { allComponents.push(dc); } }); // Legg til chips i DOM const insertPoint = componentList.querySelector('.component-chip[data-comp="wp"]').nextElementSibling; allComponents.slice(5).forEach(comp => { // Hopp over standard (de er allerede i HTML) const chip = document.createElement('span'); chip.className = 'component-chip'; chip.setAttribute('data-comp', comp.comp); chip.textContent = comp.name; componentList.insertBefore(chip, insertPoint); }); // Re-bind event listeners for alle chips document.querySelectorAll('.component-chip').forEach(chip => { chip.addEventListener('mouseenter', e => { chip.classList.add('active'); highlightComponent(chip.dataset.comp, true); }); chip.addEventListener('mouseleave', e => { if (highlightedComponent !== chip.dataset.comp) { chip.classList.remove('active'); } highlightComponent(chip.dataset.comp, false); }); chip.addEventListener('click', e => { // Fjern active fra alle document.querySelectorAll('.component-chip').forEach(c => { if (c !== chip) c.classList.remove('active'); }); // Toggle denne if (highlightedComponent === chip.dataset.comp) { chip.classList.remove('active'); highlightedComponent = null; highlightComponent(null, false); // Fjern all highlight } else { chip.classList.add('active'); highlightedComponent = chip.dataset.comp; highlightComponent(chip.dataset.comp, true); } }); }); } function highlightComponent(compName, highlight) { if (!hydro || !hydro.mesh) return; // Reset alle materialer først if (!highlight) { hydro.mesh.traverse((child) => { if (child.material) { child.material.opacity = 1.0; child.material.transparent = false; child.material.emissive?.setHex(0x000000); child.material.needsUpdate = true; } }); return; } // Highlight spesifikk komponent hydro.mesh.traverse((child) => { if (child.material) { const childName = (child.name || '').toLowerCase(); const childData = child.userData?.componentName || child.userData?.component || ''; const isTarget = child.name === compName || childData === compName || (compName === 'hull' && child === hydro.mesh) || (compName === 'wl' && child === hydro.wlContour) || (compName === 'cb' && child === hydro.cbMarker) || (compName === 'cg' && child === hydro.cgMarker) || (compName === 'wp' && child === hydro.waterPlane) || (compName === 'thrusters' && (childName.includes('thruster') || childName.includes('propeller') || childData === 'thrusters')) || (compName === 'batteries' && (childName.includes('battery') || childName.includes('batt') || childData === 'batteries')) || (compName === 'sensors' && (childName.includes('sensor') || childName.includes('lidar') || childName.includes('camera') || childData === 'sensors')) || (childName === compName.toLowerCase() || childData === compName); if (isTarget) { // Highlight denne child.material.opacity = 1.0; child.material.transparent = false; child.material.emissive?.setHex(0x444444); if (child.material.color) { child.material.color.multiplyScalar(1.5); // Gjør lysere } } else { // Fade ut andre child.material.opacity = 0.2; child.material.transparent = true; child.material.emissive?.setHex(0x000000); } child.material.needsUpdate = true; } }); } // --- Oppdater hydrostatikk-data etter lasting --- async function updateHydroAfterLoad() { // Hent ut relevante data fra hydro/core const h = hydro; const last = h.lastHydro || {}; const data = { volume: last.subV, displacement: last.subV ? last.subV * h.waterDensity / 1000 : null, // tonn cb: last.cbW, cg: last.cgW, draft: h.draftZWorld, trim: h.trimDeg, heel: h.heelDeg, gm: last.gm?.roll || null, bm: last.bm?.roll || null, wpArea: last.wp?.A || null, gz: last.gz_now?.roll || null, }; updateHydroDataUI(data); } // Kall denne etter modell er lastet og hydrostatikk er kalkulert core.onAfterRender = updateHydroAfterLoad;
